// Khai báo đường dẫn Web API lấy từ Google Apps Script const API_URL = "https://script.google.com/macros/s/AKfycbxw9_IwDZqka0NzwFKX9KihjvTP2-EEBsNGzxVBrPn4vQJMXMfwFQtsQTsliPA0xaRa/exec"; // Thay link của anh vào đây function doLogin() { const acc = document.getElementById('accInput').value.trim(); const pass = document.getElementById('passInput').value.trim(); const btn = document.getElementById('btnLogin'); const msg = document.getElementById('loginMsg'); btn.disabled = true; btn.innerHTML = 'Đang xác thực...'; msg.classList.add('hidden'); // Gửi request API dạng GET const requestUrl = `${API_URL}?action=login&account=${encodeURIComponent(acc)}&password=${encodeURIComponent(pass)}`; fetch(requestUrl) .then(response => response.json()) .then(res => { btn.disabled = false; btn.innerHTML = 'ĐĂNG NHẬP HỆ THỐNG'; if (res.status === 'success') { currentSession = res; // Cập nhật thông tin cán bộ & trường lấy từ Sheet lên giao diện Web document.getElementById('userFullName').innerText = res.user.fullName + " (" + res.user.maNhanSu + ")"; document.getElementById('userSchoolName').innerText = res.school.tenTruong; document.getElementById('userRole').innerText = res.user.vaiTro; document.getElementById('userPosition').innerText = res.user.chucVu; document.getElementById('taskbarUser').innerText = res.user.fullName; // Ẩn cửa sổ Đăng Nhập document.getElementById('loginOverlay').classList.add('hidden'); } else { msg.className = "p-3 rounded-xl text-xs font-semibold bg-rose-100 text-rose-700 border border-rose-200"; msg.innerText = "❌ " + res.message; msg.classList.remove('hidden'); } }) .catch(err => { btn.disabled = false; btn.innerHTML = 'ĐĂNG NHẬP HỆ THỐNG'; msg.className = "p-3 rounded-xl text-xs font-semibold bg-amber-100 text-amber-800 border border-amber-200"; msg.innerText = "⚠️ Lỗi kết nối API: " + err.message; msg.classList.remove('hidden'); }); }
