function doGet(e) { const action = e.parameter.action; const account = e.parameter.account; // Mã tài khoản đầy đủ (VD: C3NDT.GV001) const password = e.parameter.password; // Mật khẩu nhập vào const maTruong = e.parameter.maTruong; // Mã trường đã tách (VD: C3NDT) if (action === 'login') { return handleLogin(account, password, maTruong); } if (action === 'getHocSinh') { return handleGetHocSinh(maTruong); } } function handleLogin(account, password, maTruong) { const ss = SpreadsheetApp.getActiveSpreadsheet(); // BƯỚC 1: KIỂM TRA MÃ TRƯỜNG TRONG SHEET DSTruong const sheetDSTruong = ss.getSheetByName("DSTruong"); if (!sheetDSTruong) { return respondJSON('error', 'Không tìm thấy sheet [DSTruong] trên Google Sheet trung tâm!'); } const dataTruong = sheetDSTruong.getDataRange().getValues(); let tenTruong = "", dbUrl = ""; for (let i = 1; i < dataTruong.length; i++) { if (String(dataTruong[i][0]).trim().toUpperCase() === maTruong.toUpperCase()) { tenTruong = dataTruong[i][1]; // Cột B: Tên trường dbUrl = dataTruong[i][2]; // Cột C: Link API/CSDL break; } } if (!tenTruong) { return respondJSON('error', 'Mã trường [' + maTruong + '] không tồn tại trong danh mục!'); } // BƯỚC 2: KIỂM TRA MÃ NHÂN SỰ VÀ MẬT KHẨU TRONG SHEET TaiKhoan const sheetTaiKhoan = ss.getSheetByName("TaiKhoan"); if (!sheetTaiKhoan) { return respondJSON('error', 'Không tìm thấy sheet [TaiKhoan] trên Google Sheet trung tâm!'); } const dataTK = sheetTaiKhoan.getDataRange().getValues(); let foundUser = null; let isAccountExist = false; for (let i = 1; i < dataTK.length; i++) { let accInSheet = String(dataTK[i][0]).trim(); // Cột A: Tài khoản (VD: C3NDT.GV001) let passInSheet = String(dataTK[i][1]).trim(); // Cột B: Mật khẩu if (accInSheet.toUpperCase() === account.toUpperCase()) { isAccountExist = true; if (passInSheet === password) { foundUser = { maNhanSu: account, fullName: dataTK[i][2] || 'Cán bộ', // Cột C: Họ tên vaiTro: dataTK[i][3] || 'Giáo viên' // Cột D: Vai trò }; break; } } } // BƯỚC 3: TRẢ VỀ KẾT QUẢ CHO GIAO DIỆN OS if (!isAccountExist) { return respondJSON('error', 'Tài khoản [' + account + '] không tồn tại!'); } if (!foundUser) { return respondJSON('error', 'Mật khẩu không chính xác! Vui lòng thử lại.'); } // Đăng nhập thành công return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'Đăng nhập thành công!', user: foundUser, school: { maTruong: maTruong, tenTruong: tenTruong, dbUrl: dbUrl } })).setMimeType(ContentService.MimeType.JSON); } function respondJSON(status, message) { return ContentService.createTextOutput(JSON.stringify({ status: status, message: message })).setMimeType(ContentService.MimeType.JSON); }
